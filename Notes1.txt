  if (user) {

      const { payload, events } = response // payload is record, events is array of string describing event (added, deleted, modified etc)
      if (events[0].includes('create')) {
        setBooks((prevBooks) => [...prevBooks, payload])

  }

// NEW VERSION
import {..., realtime} from "../lib/appwrite"
  
useEffect(() => {
    let subscription;
    let isMounted = true;

    if (user) {
      fetchBooks();

      const setupRealtime = async () => {
        const sub = await realtime.subscribe(
          Channel.tablesdb(DATABASE_ID).table(TABLE_ID).row(),
          (response) => {
            const { payload, events } = response;
            console.log("response in channel :>> ", response);
            if (events[0].includes("create")) {
              setBooks((prevBooks) => [...prevBooks, payload]);
            }
          }
        );

        if (!isMounted) {
          await sub.unsubscribe();
        } else {
          subscription = sub;
        }
      };
      setupRealtime();
    } else {
      setBooks([]);
    }

    return () => {
      isMounted = false;
      if (subscription) subscription.unsubscribe();
    };
  }, [user]);
